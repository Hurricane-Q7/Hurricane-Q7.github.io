4.3-C语言-函数指针、函数指针数组、指向函数指针数组的指针

2026-04-02 4 min read # C语言学习笔记

函数指针

函数指针是一种指针，是指向函数的指针，就像数组指针是指向数组的指针！
函数名就是函数的地址， 也可以通过&函数名来得到函数的地址！

int test(const char* str)
{
    return 0;
}
int main()
{
    //创建函数指针变量！
   int (*pf)(const char*) = test//or &test
}

函数指针的用法；
得到函数的地址后，我们可以通过该地址来间接调用函数！
int test(const char* str)
{
	printf("test()\n");
	return 0;
}
int main()
{
	printf("%p\n", test);
	printf("%p\n", &test);//test和&test是等效的，都表示函数的地址
	int (*pf)(const char*) = test;//定义指向test函数的指针变量pf，pf存放函数test的地址
	(*pf)("abc");
	pf("abc");//pf 与 *pf两个是一样的，里面都是函数的地址，因为函数的函数名就是函数的地址
	test("abc");
	return 0;
}

函数指针的用法（简易计算器）

//计算器第二版，用函数指针的知识来简化代码
#define _CRT_SECURE_NO_WARNINGS
#include<stdio.h>
//先实现一个简易版计算器（支持整数的加减乘除运算）
void menu()
{
	printf("*************************\n");
	printf("****1. add 2. sub *******\n");
	printf("****3. mul 4. div *******\n");
	printf("****0.  exit      *******\n");
}
int add(int x, int y)
{
	return x + y;
}
int sub(int x, int y)
{
	return x - y;
}
int mul(int x, int y)
{
	return x * y;
}
int div(int x, int y)
{
	return x / y;
}
int cal(int (*pf)(int, int))
{
	int x = 0;
	int y = 0;
	int ret = 0;
	printf("请输入两个数字：");
	scanf("%d %d", &x, &y);
	ret = (*pf)(x, y);
	printf("结果是：%d", ret);
}
int main()
{
	int cal(int (*pf)(int, int));
	int input = 0;
	menu();
	printf("请输入你的选项:>");
	scanf("%d", &input);
	switch (input)
	{
	case 1:
		cal(add);//把函数的地址给传过去，来实现对add函数的间接调用
		break;
	case 2:
		cal(sub);//同上
		break;
	case 3:
		cal(mul);//同上
		break;
	case 4:
		cal(div);//同上
		break;
	case 0:
		printf("程序退出！\n");
		break;//这个break是用来跳出switch语句的
		break;//这个break是用来当作case语句的终点的，来阻止case语句往下穿透！
	default:
		printf("输入错误！"); 
		break;
	}
	return 0;
}

函数指针数组

函数指针是一种指针，是指向一个函数的，就像数组指针是指向一个数组的指针！
指针可以被数组存放起来，也就是指针数组，那么函数指针被数组存放起来，也就是函数指针数组！

#define _CRT_SECURE_NO_WARNINGS
#include<stdio.h>
int add(int x, int y)
{
	return x + y;
}
int sub(int x, int y)
{
	return x - y;
}
int mul(int x, int y)
{
	return x * y;
}
int div(int x, int y)
{
	return x / y;
}
int main()
{
	int (*arr[4])(int, int) = {add,sub,mul,div};//函数指针数组，用来存放函数指针的一个数组
	int i = 0;
	for (i = 0; i < 4; i++)
	{
		int ret = arr[i](8, 4);
		printf("%d \n", ret);
	}
	return 0;
}
