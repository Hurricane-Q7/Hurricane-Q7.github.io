//目标数字生成要求：

//2. 一共生成4位数字！
//3. 不能有重复的数字！
void generate_target(int target[])//目标数字生成器
{
 int num = 0;//num代表生成的数字
 int i = 0;//用i来代表下标位数
 int used[10] = { 0 };//用一个数组来标记生成的数字有没有用过，用过标记为1，没有用过标记为0
 //第一位数字生成
 while (i == 0)
 {
 	int num = rand() % 10;//用一个变量来存放生成的数字（0~9）;
 	if (num != 0 && !used[num])//!1为假，也就是代表用过了，!0为真，代表没有被标记，可以使用！
 	{
 		target[i++] = num;//赋值完即使更新调整
 		used[num] = 1;
 	}
 }
 //后三位数字生成
 while (i < 4)
 {
 	int num = rand() % 10;
 	if (!used[num])//后三位只要满足不重复即可
 	{
 		target[i++] = num;//当前i位置，数字赋值完再进行下一位
 		used[num] = 1;
 	}
 }

}

  1. 反馈机制（计算A与B）
void calculate_ab(int target[], int guess[], int* a, int* b)
{
 //计算a与b
 *a = 0;
 *b = 0;
 int i = 0;
 int target_used[4] = { 0 };
 int guess_used[4] = { 0 };
 for (i = 0; i < 4; i++)
 {
 	if (guess[i] == target[i])
 	{
 		(*a)++;
 		target_used[i] = 1;
 		guess_used[i] = 1;
 	}
 }
 for (i = 0; i < 4; i++)
 {
 	if (guess_used[i] == 0)//先排查已经被计算过A的元素
 	{
 		for (int j = 0; j < 4; j++)
 		{
 			if (!target_used[j] && target[j] == guess[i])
 			{
 				(*b)++;
 				target_used[j] = 1;
 				break;
 			}
 		}
 	}
 }
}

break 与 continue在循环中的差异

简单来说：break 是直接终止整个循环，而 continue 是跳过当前次循环的剩余代码，直接进入下一次循环判断部分，或循环控制部分（for循环）

#include <stdio.h>

int main() {
    // 循环打印 1-5，但遇到 3 就跳过本次打印，继续下一次循环
    for (int i = 1; i <= 5; i++) {
        if (i == 3) {
            continue;  // 触发 continue，跳过本次循环的 printf，直接去判断 i++
        }
        printf("i = %d\n", i);  // continue 后这行代码本次不执行，但下一次循环会执行
    }
    printf("循环结束\n");
    return 0;
}

常见用法
break：常用于「满足某个条件时提前结束循环」，比如：遍历数组找到目标值后立即停止遍历、用户输入错误时退出输入循环。
continue：常用于「过滤掉不需要处理的情况」，比如：遍历数字时跳过偶数、读取文件时跳过空行。
总结

  • break：终止整个循环，循环彻底结束，程序执行循环外的代码；
  • continue：while与dowhilez中直接跳过剩余代码，来到循环条件判断部分，for循环中来到了循环变量控制部分（表达式3）
    注意：while/do-while 中用 continue 需手动更新循环变量，避免无限循环。
#include <stdio.h>

int main() {
    int i = 1;  // 相当于for的表达式1
    while (i <= 5) {  // 相当于for的表达式2
        if (i == 3) {
            printf("触发continue\n");
            // 如果这里不手动i++，会无限循环！
            // i++; 
            continue;  // 直接跳回while条件判断（i<=5），不会自动更新i
        }
        printf("当前i = %d\n", i);
        i++;  // 相当于for的表达式3
    }
    return 0;
}

第一次重写版错误代码1

错误原因i++,位置错误与使用逻辑错误，应该等生成的数字（后三位）检验玩后，i再作调整，这样则一直再为当前target[i]找合适的数字进行保存，则不会跳到下一位，而下面的代码就是我犯的错误，按照我的代码如果该数字重复，则执行i++，条过当前位置i,为i+1寻找数字！
void generate_target(int target[])
{
	int i = 0;
	int used[10] = { 0 };//每次调用generate_target都会清空used数组信息，避免下次影响使用！
	
	while (i == 0)
	{
		int num = rand() % 10;
		if (num!= 0&&!used[num])
		{
			target[i++] = num;
			used[num] = 1;
		}
	}
	while (i < 4)
	{
		int num = rand() % 10;

计算AB

总结：写代码前要明确实现的功能也就是代码起到的作用，最重要的是要先用自然语言把逻辑讲清楚，把逻辑拆解到不能再拆解的样子最好，然后再结合语法知识用代码实现具体的功能！

void calculate_ab(int target[], int guess[], int* a, int* b)
{
	*a = 0;
	*b = 0;
	int i = 0;
	int j = 0;
	int target_used[4] = { 0 };//定义两个标记型数组，用来标记数组元素是否已经统计过A或B了，0:未统计,1:已统计
	int guess_used[4] = { 0 };//
	for (i = 0; i < 4; i++)//计算A的数量(A代表正确的数字在正确的位置上)
	{                      //计算逻辑：只需要将guess与target中的元素一一比较即可，相等就加一，不相等就继续
		if (guess[i] == target[i])
		{
			(*a)++;
			target_used[i] = 1;//对统计过的数组元素进行标记
			guess_used[i] = 1;
		}
	}
	//i 代表target的索引
	//j 代表guess的索引
	for (i = 0; i < 4; i++)//计算B(B代表正确的数字在错误的位置上)
	{                      //计算逻辑：遍历target中「没算过A」的位，去guess里找「没算过A」且数字相同的位
		if (!target_used[i])
		{
			for (j = 0; j < 4; j++)
			{
				if (!guess_used[j] && target[i] == guess[j])
				{
					(*b)++;
					guess_used[j] = 1;
					break;
				}
			}
		}
	}
	printf("反馈：%dA%dB\n", *a, *b);
}
//计算逻辑：
//先计算A（数字对、位置对）的值，再计算B（数字对、位置不对）！
//计算A：只需要讲target与gues中的元素一一比较即可，相等A就加一，不相等就找下一位！
//计算B: 首先要知道，已经算过A的数字肯定不能再参与计算B了，不然就会造成重复，因此要筛选已经用
//过的数字，核心逻辑是，从没有被使用过的target数组元素里面找，然后在从guess里面没有用过的元素（隔绝相同的位置）找target与guess相等的元素，找到后B就加1然后再对guess数组进行标记，找不到就下一位！
